BEGIN

        s.SalaryDays,
        s.Arrears,
        s.OtherDeduction,
        b.BranchName,
		dp.DepartmentName,
		d.DesignationName,
		e.DateOfBirth,

        esa.ChildEducationAllowance AS ChildEducationAllowanceRate,

          (@MonthNumber = 0 AND @Year = 0) OR

      )
      AND (
          (@EmployeeCodes = '' OR s.EmployeeCode IN (SELECT value FROM STRING_SPLIT(@EmployeeCodes, ',')))
      )
      AND (
          (@BranchId = 0 OR e.BranchId = @BranchId)
      )
END

Create  PROCEDURE [dbo].[sp_GetEmployeePayableDaysWithMonthDays]
    @StartDate DATE,
    @EndDate DATE,
    @EmployeeIds NVARCHAR(MAX) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    -- Calculate total days in the month (date range)
    DECLARE @TotalMonthDays INT = DATEDIFF(DAY, @StartDate, @EndDate) + 1;

    -- Temp table to store attendance data
    IF OBJECT_ID('tempdb..#AttendanceData') IS NOT NULL DROP TABLE #AttendanceData;

    CREATE TABLE #AttendanceData (
        BranchName NVARCHAR(200),
        EmployeeCode NVARCHAR(50),
        FullName NVARCHAR(200),
        Emp_Id INT,
        PayableDay DECIMAL(10,2)
    );

    -- Get only the required data (Employee details + PayableDay)
    INSERT INTO #AttendanceData
    SELECT
        BranchName,
        EmployeeCode,
        FullName,
        Emp_Id,
        PayableDay
    FROM dbo.fn_GetEmployeeAttendanceStatus_V2(@StartDate, @EndDate, @EmployeeIds)
    OPTION (RECOMPILE);

    -- Final result: Employee details + TotalPayableDays + TotalMonthDays
    SELECT
        BranchName,
        EmployeeCode,
        FullName,
        Emp_Id,
        SUM(PayableDay) AS TotalPayableDays,
        @TotalMonthDays AS TotalMonthDays
    FROM #AttendanceData
    GROUP BY BranchName, EmployeeCode, FullName, Emp_Id
    ORDER BY BranchName, EmployeeCode, FullName;

    -- Clean up
    DROP TABLE #AttendanceData;
END

USE [HRMS_Pro]
GO
/****** Object:  StoredProcedure [dbo].[USP_CalculateMonthlySalary_V2]    Script Date: 02-05-2026 07:20:48 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON

ALTER PROCEDURE [dbo].[USP_CalculateMonthlySalary_V2]
    @StartDate DATE,
    @EndDate DATE,
    @CompanyId INT = 0,
    @EmployeeCodes VARCHAR(MAX) = NULL,
    @BranchIds VARCHAR(MAX) = NULL,
    @Action VARCHAR(50) = 'GetData',
    @CreatedBy INT = NULL
AS
BEGIN
    SET NOCOUNT ON;
    TRUNCATE TABLE dbo.AttendanceSummaryTemp;

    DECLARE @Month INT = MONTH(@StartDate);
    DECLARE @Year INT = YEAR(@StartDate);
    DECLARE @ProcessingDate DATE = @StartDate;

    -- Employee Code filtering
    DECLARE @EmployeeCodeTable TABLE (EmployeeCode NVARCHAR(50));
    IF @EmployeeCodes IS NOT NULL
    BEGIN
        INSERT INTO @EmployeeCodeTable SELECT TRIM(value) FROM STRING_SPLIT(@EmployeeCodes, ',') WHERE TRIM(value) <> '';
    END;

    -- Branch filtering
    DECLARE @BranchIdTable TABLE (BranchId INT);
    IF @BranchIds IS NOT NULL
    BEGIN
        INSERT INTO @BranchIdTable SELECT CAST(TRIM(value) AS INT) FROM STRING_SPLIT(@BranchIds, ',') WHERE TRIM(value) <> '' AND ISNUMERIC(TRIM(value)) = 1;
    END;

    IF @Action IN ('GetData','Insert')
    BEGIN
        -- Get Employee IDs
        DECLARE @EmployeeIdsList NVARCHAR(MAX);
        SELECT @EmployeeIdsList = STRING_AGG(CAST(U.Id AS NVARCHAR(10)), ',')
        FROM AspNetUsers U WITH (NOLOCK)
        WHERE U.IsEnabled = 1 AND U.IsDeleted = 0 AND U.IsBlocked = 0
          AND (@CompanyId = 0 OR ISNULL(U.CompanyId, 0) = @CompanyId)
          AND (@BranchIds IS NULL OR U.BranchId IN (SELECT BranchId FROM @BranchIdTable))
          AND (@EmployeeCodes IS NULL OR U.EmployeeCode IN (SELECT EmployeeCode FROM @EmployeeCodeTable));

        -- Get Attendance Data from NEW Function V2
        CREATE TABLE #AttendanceData (
            Emp_Id INT,
            CalendarDate DATE,
            Status NVARCHAR(10),
            WorkingHours DECIMAL(5,2),
            PresentDays DECIMAL(10,2),
            AbsentDays DECIMAL(10,2),
            WeekOffDays DECIMAL(10,2),
            LeaveDays DECIMAL(10,2),
            HolidayDays DECIMAL(10,2),
            HalfDayHDDays DECIMAL(10,2),
            LWPDays DECIMAL(10,2),
            CompOffDays DECIMAL(10,2),
            PayableDay DECIMAL(10,2),
            UnpaidDay DECIMAL(10,2),
            SalaryDay DECIMAL(10,2),
            INDEX IX_AttendanceData_EmpId (Emp_Id)
        );

        INSERT INTO #AttendanceData
        SELECT Emp_Id, CalendarDate, Status, WorkingHours, PresentDays, AbsentDays, WeekOffDays, LeaveDays, 
               HolidayDays, HalfDayHDDays, LWPDays, CompOffDays, PayableDay, UnpaidDay, SalaryDay
        FROM dbo.fn_GetEmployeeAttendanceStatus_V2(@StartDate, @EndDate, @EmployeeIdsList);

        -- Get Employee Master, Earning, and Deduction Data
        CREATE TABLE #Employees (EmployeeId INT PRIMARY KEY, GrossSalary DECIMAL(10,2), BasicSalary DECIMAL(10,2), IsPFApplicable BIT, EmployeeCode NVARCHAR(50), BranchId INT, EmployeeName NVARCHAR(255), CompanyId INT);
        
        INSERT INTO #Employees
        SELECT DISTINCT U.Id, ISNULL(U.GrossSalary, 0), ISNULL(U.BasicSalary, 0), ISNULL(U.IsPFApplicable, 0), U.EmployeeCode, U.BranchId, U.FullName, ISNULL(U.CompanyId, 0)
        FROM AspNetUsers U WITH (NOLOCK)
        WHERE U.IsEnabled = 1 AND U.IsDeleted = 0 AND U.IsBlocked = 0
          AND (@CompanyId = 0 OR ISNULL(U.CompanyId, 0) = @CompanyId)
          AND (@BranchIds IS NULL OR U.BranchId IN (SELECT BranchId FROM @BranchIdTable))
          AND (@EmployeeCodes IS NULL OR U.EmployeeCode IN (SELECT EmployeeCode FROM @EmployeeCodeTable));

        CREATE TABLE #EarningData (EmployeeId INT PRIMARY KEY, Basic DECIMAL(10,2), HRA DECIMAL(10,2), Conveyance DECIMAL(10,2), ChildEducationAllowance DECIMAL(10,2), Medical DECIMAL(10,2), Deputation DECIMAL(10,2), Arrears DECIMAL(10,2));
        INSERT INTO #EarningData SELECT EmployeeId, Basic, HRA, Conveyance, ChildEducationAllowance, Medical, Deputation, Arrears FROM Earning WITH (NOLOCK) WHERE IsEnabled = 1 AND Month = @Month AND Year = @Year AND EmployeeId IN (SELECT EmployeeId FROM #Employees);

        CREATE TABLE #DeductionData (EmployeeId INT PRIMARY KEY, LWF DECIMAL(10,2), TDS DECIMAL(10,2), Loan DECIMAL(10,2), OtherDeduction DECIMAL(10,2));
        INSERT INTO #DeductionData SELECT EmployeeId, LWF, TDS, Loan, OtherDeduction FROM Deduction WITH (NOLOCK) WHERE IsEnabled = 1 AND Month = @Month AND Year = @Year AND EmployeeId IN (SELECT EmployeeId FROM #Employees);

        -- Calculate Attendance Summary using Function V2 breakdown columns
        CREATE TABLE #AttendanceSummary (
            EmployeeId INT PRIMARY KEY, EmployeeCode NVARCHAR(50), EmployeeName NVARCHAR(255), GrossSalary DECIMAL(10,2), BasicSalary DECIMAL(10,2), 
            IsPFApplicable BIT, CompanyId INT, TotalP DECIMAL(10,2), TotalA DECIMAL(10,2), TotalW DECIMAL(10,2), TotalL DECIMAL(10,2), TotalH DECIMAL(10,2), 
            TotalHF DECIMAL(10,2), TotalCO DECIMAL(10,2), TotalLWP DECIMAL(10,2), PayableDays DECIMAL(10,2), UnpaidDays DECIMAL(10,2), MonthDays INT, 
            SalaryDays DECIMAL(10,2), PresentCount INT, AbsentCount INT, HolidayCount INT, WeekOffCount INT, LeaveCount INT, HalfDayCount INT, LWPCount INT
        );

        INSERT INTO #AttendanceSummary
        SELECT E.EmployeeId, E.EmployeeCode, E.EmployeeName, E.GrossSalary, E.BasicSalary, E.IsPFApplicable, E.CompanyId,
               ISNULL(SUM(AD.PresentDays), 0) AS TotalP, 
               ISNULL(SUM(AD.AbsentDays), 0) AS TotalA,                          -- FIX: removed duplicate + LWPDays to prevent double-counting
               ISNULL(SUM(AD.WeekOffDays), 0) AS TotalW, 
               ISNULL(SUM(AD.LeaveDays), 0) AS TotalL,
               ISNULL(SUM(AD.HolidayDays), 0) AS TotalH, 
               ISNULL(SUM(AD.HalfDayHDDays), 0) AS TotalHF, 
               ISNULL(SUM(AD.CompOffDays), 0) AS TotalCO, 
               ISNULL(SUM(AD.LWPDays), 0) AS TotalLWP,
               ISNULL(SUM(AD.PayableDay), 0) AS PayableDays, 
               ISNULL(SUM(AD.UnpaidDay), 0) AS UnpaidDays, 
               DATEDIFF(DAY, @StartDate, @EndDate) + 1 AS MonthDays,
               ISNULL(SUM(AD.SalaryDay), 0) AS SalaryDays,
               ISNULL(COUNT(CASE WHEN AD.Status IN ('P', 'P*', 'P^') THEN 1 END), 0) AS PresentCount,
               ISNULL(COUNT(CASE WHEN AD.Status IN ('A', 'HCO') THEN 1 END), 0) AS AbsentCount,
               ISNULL(COUNT(CASE WHEN AD.Status = 'H' THEN 1 END), 0) AS HolidayCount,
               ISNULL(COUNT(CASE WHEN AD.Status = 'W' THEN 1 END), 0) AS WeekOffCount,
               ISNULL(COUNT(CASE WHEN AD.Status IN ('PL', 'HPL', 'CO', 'CO*', 'CO^') THEN 1 END), 0) AS LeaveCount,
               ISNULL(COUNT(CASE WHEN AD.Status IN ('HF', 'HD', 'P^') THEN 1 END), 0) AS HalfDayCount,
               ISNULL(COUNT(CASE WHEN AD.Status = 'LWP' THEN 1 END), 0) AS LWPCount
        FROM #Employees E
        LEFT JOIN #AttendanceData AD ON E.EmployeeId = AD.Emp_Id AND AD.Status <> '-'
        GROUP BY E.EmployeeId, E.EmployeeCode, E.EmployeeName, E.GrossSalary, E.BasicSalary, E.IsPFApplicable, E.CompanyId;

        -- ============================================================
        -- HR PAYABLE DAYS OVERRIDE
        -- ============================================================
        UPDATE ASM
        SET
            ASM.PayableDays = OVR.OverridePayableDays,
            ASM.SalaryDays  = OVR.OverridePayableDays,
            ASM.UnpaidDays  = ASM.MonthDays - OVR.OverridePayableDays
        FROM #AttendanceSummary ASM
        INNER JOIN dbo.SalaryPayableDaysOverride OVR
            ON  ASM.EmployeeId   = OVR.EmployeeId
            AND OVR.MonthNumber  = @Month
            AND OVR.Year         = @Year
            AND OVR.IsEnabled    = 1
            AND OVR.IsDeleted    = 0;

        -- ============================================================
        -- HR SPECIAL OVERRIDE LIST
        -- ============================================================
        DECLARE @ZeroAllowanceEmployees TABLE (EmployeeId INT);
        INSERT INTO @ZeroAllowanceEmployees (EmployeeId) VALUES (165), (218);

        -- Calculate Salary with CTE for cleaner code
        ;WITH SalaryComponents AS (
            SELECT 
                ASM.EmployeeId, ASM.EmployeeCode, ASM.EmployeeName, ASM.GrossSalary, ASM.CompanyId, ASM.IsPFApplicable,
                ASM.PayableDays, ASM.MonthDays, ASM.HolidayCount, ASM.TotalA, ASM.TotalP, ASM.WeekOffCount, 
                ASM.TotalL + ASM.TotalHF AS TotalLeave, 
                ASM.SalaryDays, ASM.HalfDayCount, ASM.LWPCount,
                @Month AS MonthNumber, 
                DATENAME(MONTH, DATEFROMPARTS(@Year, @Month, 1)) AS MonthName, 
                @Year AS Year,
                
                -- Pre-calculate proration factor
                CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END AS ProrationFactor,
                
                -- Pre-calculate Basic Salary (earned)
                CASE 
                    WHEN ASM.CompanyId = 11 THEN ROUND(ASM.GrossSalary * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                    WHEN ISNULL(E.Basic, 0) > 0 THEN E.Basic 
                    WHEN ASM.BasicSalary > 0 THEN ROUND(ASM.BasicSalary * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                    ELSE ROUND((ASM.GrossSalary * 0.40) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                END AS EarnedBasic,
                
                -- *** HRA: 0 for special override employees ***
                CASE 
                    WHEN ASM.CompanyId = 11 THEN 0
                    WHEN ASM.EmployeeId IN (SELECT EmployeeId FROM @ZeroAllowanceEmployees) THEN 0  -- HR OVERRIDE
                    WHEN ISNULL(E.HRA, 0) > 0 THEN E.HRA 
                    WHEN ASM.BasicSalary > 0 THEN ROUND((ASM.BasicSalary * 0.40) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                    ELSE ROUND(((ASM.GrossSalary * 0.40) * 0.40) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                END AS EarnedHRA,
                
                -- *** Conveyance: 0 for special override employees ***
                CASE 
                    WHEN ASM.CompanyId = 11 THEN 0
                    WHEN ASM.EmployeeId IN (SELECT EmployeeId FROM @ZeroAllowanceEmployees) THEN 0  -- HR OVERRIDE
                    WHEN ISNULL(E.Conveyance, 0) > 0 THEN E.Conveyance 
                    ELSE ROUND(1600 * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                END AS EarnedConveyance,
                
                -- *** Child Education Allowance: 0 for special override employees ***
                CASE 
                    WHEN ASM.CompanyId = 11 THEN 0
                    WHEN ASM.EmployeeId IN (SELECT EmployeeId FROM @ZeroAllowanceEmployees) THEN 0  -- HR OVERRIDE
                    WHEN ISNULL(E.ChildEducationAllowance, 0) > 0 THEN E.ChildEducationAllowance 
                    ELSE ROUND(200 * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                END AS EarnedChildEdu,
                
                -- *** Medical Allowance: 0 for special override employees ***
                CASE 
                    WHEN ASM.CompanyId = 11 THEN 0
                    WHEN ASM.EmployeeId IN (SELECT EmployeeId FROM @ZeroAllowanceEmployees) THEN 0  -- HR OVERRIDE
                    WHEN ISNULL(E.Medical, 0) > 0 THEN E.Medical 
                    ELSE ROUND(1250 * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                END AS EarnedMedical,
                
                -- *** Deputation: For override employees = GrossSalary - BasicSalary (since HRA/Conv/ChildEdu/Medical all = 0) ***
                -- For normal employees: GrossSalary - Basic - HRA - Conveyance - ChildEdu - Medical
                CASE 
                    WHEN ASM.CompanyId = 11 THEN 0
                    WHEN ASM.EmployeeId IN (SELECT EmployeeId FROM @ZeroAllowanceEmployees) THEN
                        -- HR OVERRIDE: Deputation = Gross - Basic (prorated), all other allowances are 0
                        CASE 
                            WHEN ISNULL(E.Deputation, 0) > 0 THEN E.Deputation
                            WHEN ASM.BasicSalary > 0 THEN ROUND((ASM.GrossSalary - ASM.BasicSalary) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                            ELSE ROUND((ASM.GrossSalary - (ASM.GrossSalary * 0.40)) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                        END
                    WHEN ISNULL(E.Deputation, 0) > 0 THEN E.Deputation 
                    WHEN ASM.BasicSalary > 0 THEN ROUND((ASM.GrossSalary - ASM.BasicSalary - (ASM.BasicSalary * 0.40) - 200 - 1600 - 1250) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                    ELSE ROUND((ASM.GrossSalary - (ASM.GrossSalary * 0.40) - 200 - 1600 - ((ASM.GrossSalary * 0.40) * 0.40) - 1250) * (CASE WHEN ASM.MonthDays > 0 THEN ASM.PayableDays / CAST(ASM.MonthDays AS DECIMAL(10,4)) ELSE 0 END), 2)
                END AS EarnedDeputation,
                
                -- Arrears
                CASE WHEN ASM.PayableDays = 0 THEN 0 ELSE ROUND(ISNULL(E.Arrears, 0), 2) END AS Arrears,
                
                -- Deductions from master
                CASE WHEN ASM.PayableDays = 0 THEN 0 ELSE ROUND(ISNULL(D.LWF, 0), 2) END AS LWF,
                CASE WHEN ASM.PayableDays = 0 THEN 0 ELSE ROUND(ISNULL(D.TDS, 0), 2) END AS TDS,
                CASE WHEN ASM.PayableDays = 0 THEN 0 ELSE ROUND(ISNULL(D.Loan, 0), 2) END AS Loan,
                CASE WHEN ASM.PayableDays = 0 THEN 0 ELSE ROUND(ISNULL(D.OtherDeduction, 0), 2) END AS OtherDeduction
                
            FROM #AttendanceSummary ASM
            LEFT JOIN #EarningData E ON ASM.EmployeeId = E.EmployeeId
            LEFT JOIN #DeductionData D ON ASM.EmployeeId = D.EmployeeId
        )
        SELECT
            SC.EmployeeId, SC.EmployeeCode, SC.EmployeeName, SC.GrossSalary,
            SC.PayableDays, SC.MonthDays, SC.HolidayCount AS Holiday, SC.TotalA AS AbsentDays, SC.TotalP AS PresentDays,
            SC.WeekOffCount AS WeekOff, 
            SC.TotalLeave AS Leave, 
            SC.SalaryDays, SC.HalfDayCount AS HalfDays, SC.LWPCount AS LWPDays,
            SC.MonthNumber, SC.MonthName, SC.Year,
  
            -- Earnings
            CAST(SC.EarnedBasic AS DECIMAL(10,2)) AS BasicSalary,
            CAST(SC.EarnedHRA AS DECIMAL(10,2)) AS HRA,
            CAST(SC.EarnedConveyance AS DECIMAL(10,2)) AS ConveyanceAllowance,
            CAST(SC.EarnedChildEdu AS DECIMAL(10,2)) AS ChildEducationAllowance,
            CAST(SC.EarnedMedical AS DECIMAL(10,2)) AS MedicalAllowance,
            CAST(SC.EarnedDeputation AS DECIMAL(10,2)) AS DeputationAllowance,
            CAST(SC.Arrears AS DECIMAL(10,2)) AS Arrears,
            
            -- Total Gross Salary
            CAST(SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears AS DECIMAL(10,2)) AS TotalGrossSalary,
            
            -- PF (12% of Basic, capped at 1800)
            CAST(CASE 
                WHEN SC.PayableDays = 0 OR SC.IsPFApplicable = 0 OR SC.CompanyId = 11 THEN 0
                WHEN (SC.EarnedBasic * 0.12) > 1800 THEN 1800 
                ELSE ROUND(SC.EarnedBasic * 0.12, 0) 
            END AS DECIMAL(10,2)) AS PF,
            
            -- ESIC (0.75% of Total Gross if base GrossSalary <= 21000)
            CAST(CASE 
                WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 OR SC.GrossSalary > 21000 THEN 0
                ELSE ROUND((SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) * 0.0075, 0)
            END AS DECIMAL(10,2)) AS ESIC,
            
            -- Professional Tax (200 if earned gross > 12000)
            -- FIX: Added SC.CompanyId = 11 exclusion (was missing, causing ?200 mismatch for Company 11 employees)
            CAST(CASE 
                WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 THEN 0
                WHEN (SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) > 12000 THEN 200 
                ELSE 0 
            END AS DECIMAL(10,2)) AS ProfessionalTax,
		            
            -- Group Medical (266 if base GrossSalary >= 21000)
            CAST(CASE 
                WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 THEN 0
                WHEN SC.GrossSalary >= 21000 THEN 266 
                ELSE 0 
            END AS DECIMAL(10,2)) AS GroupMedical,
            
            -- Term Insurance (from function)
            CAST(CASE 
                WHEN SC.PayableDays = 0 THEN 0
                ELSE dbo.fn_GetTermInsuranceDeduction(SC.GrossSalary, @ProcessingDate, SC.CompanyId) 
            END AS DECIMAL(10,2)) AS TermInsurance,
            
            -- Other Deductions
            CAST(SC.LWF AS DECIMAL(10,2)) AS LWF,
            CAST(SC.TDS AS DECIMAL(10,2)) AS TDS,
            CAST(SC.Loan AS DECIMAL(10,2)) AS Loan,
            CAST(SC.OtherDeduction AS DECIMAL(10,2)) AS OtherDeduction,
            
            -- Total Deductions
            -- FIX: ProfessionalTax inline block now includes SC.CompanyId = 11 exclusion (was missing before)
            CAST(
                -- PF
                CASE WHEN SC.PayableDays = 0 OR SC.IsPFApplicable = 0 OR SC.CompanyId = 11 THEN 0 WHEN (SC.EarnedBasic * 0.12) > 1800 THEN 1800 ELSE ROUND(SC.EarnedBasic * 0.12, 0) END +
                -- ESIC
                CASE WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 OR SC.GrossSalary > 21000 THEN 0 ELSE ROUND((SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) * 0.0075, 0) END +
                -- Professional Tax -- FIX: added OR SC.CompanyId = 11
                CASE WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 THEN 0 WHEN (SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) > 12000 THEN 200 ELSE 0 END +
                -- Group Medical
                CASE WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 THEN 0 WHEN SC.GrossSalary >= 21000 THEN 266 ELSE 0 END +
                -- Term Insurance
                CASE WHEN SC.PayableDays = 0 THEN 0 ELSE dbo.fn_GetTermInsuranceDeduction(SC.GrossSalary, @ProcessingDate, SC.CompanyId) END +
                SC.LWF + SC.TDS + SC.Loan + SC.OtherDeduction
            AS DECIMAL(10,2)) AS TotalDeductions,
            
            -- Net Salary
            -- FIX: ProfessionalTax inline block now includes SC.CompanyId = 11 exclusion (was missing before)
            CAST(
                (SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) -
                (
                    -- PF
                    CASE WHEN SC.PayableDays = 0 OR SC.IsPFApplicable = 0 OR SC.CompanyId = 11 THEN 0 WHEN (SC.EarnedBasic * 0.12) > 1800 THEN 1800 ELSE ROUND(SC.EarnedBasic * 0.12, 0) END +
                    -- ESIC
                    CASE WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 OR SC.GrossSalary > 21000 THEN 0 ELSE ROUND((SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) * 0.0075, 0) END +
                    -- Professional Tax -- FIX: added OR SC.CompanyId = 11
                    CASE WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 THEN 0 WHEN (SC.EarnedBasic + SC.EarnedHRA + SC.EarnedConveyance + SC.EarnedChildEdu + SC.EarnedMedical + SC.EarnedDeputation + SC.Arrears) > 12000 THEN 200 ELSE 0 END +
                    -- Group Medical
                    CASE WHEN SC.PayableDays = 0 OR SC.CompanyId = 11 THEN 0 WHEN SC.GrossSalary >= 21000 THEN 266 ELSE 0 END +
                    -- Term Insurance
                    CASE WHEN SC.PayableDays = 0 THEN 0 ELSE dbo.fn_GetTermInsuranceDeduction(SC.GrossSalary, @ProcessingDate, SC.CompanyId) END +
                    SC.LWF + SC.TDS + SC.Loan + SC.OtherDeduction
                )
            AS DECIMAL(10,2)) AS NetSalary

        INTO #FinalSalaryDetails
        FROM SalaryComponents SC;

        -- Handle Actions
        IF @Action = 'GetData' OR @Action IS NULL
        BEGIN
            -- FIX: Scoped DELETE to current Month+Year only to prevent wiping other months' temp data
            DELETE FROM dbo.AttendanceSummaryTemp
            WHERE MonthNumber = @Month AND Year = @Year;
        
            INSERT INTO dbo.AttendanceSummaryTemp (
                EmployeeId, EmployeeCode, EmployeeName, GrossSalary, PayableDays, MonthDays, Holiday, AbsentDays, PresentDays, 
                WeekOff, Leave, SalaryDays, MonthNumber, MonthName, Year, BasicSalary, HRA, ConveyanceAllowance, ChildEducationAllowance, 
                MedicalAllowance, DeputationAllowance, Arrears, TotalGrossSalary, PF, ESIC, ProfessionalTax, GroupMedical, TermInsurance, 
                LWF, TDS, Loan, OtherDeduction, TotalDeductions, NetSalary
            )
            SELECT EmployeeId, EmployeeCode, EmployeeName, GrossSalary, PayableDays, MonthDays, Holiday, AbsentDays, PresentDays,
                   WeekOff, Leave, SalaryDays, MonthNumber, MonthName, Year, BasicSalary, HRA, ConveyanceAllowance, ChildEducationAllowance,
                   MedicalAllowance, DeputationAllowance, Arrears, TotalGrossSalary, PF, ESIC, ProfessionalTax, GroupMedical, TermInsurance,
                   LWF, TDS, Loan, OtherDeduction, TotalDeductions, NetSalary
            FROM #FinalSalaryDetails;
        
            -- Check if any records already exist
            DECLARE @ExistingCount INT;
            DECLARE @NewCount INT;
            
            SELECT @ExistingCount = COUNT(*)
            FROM #FinalSalaryDetails FSD
            WHERE EXISTS (
                SELECT 1 FROM SalaryDetails SD WITH (NOLOCK)
                WHERE SD.EmployeeId = FSD.EmployeeId 
                AND SD.MonthNumber = @Month 
                AND SD.Year = @Year 
                AND SD.isdeleted = 0 
                AND SD.IsEnabled = 1
            );
            
            SELECT @NewCount = COUNT(*)
            FROM #FinalSalaryDetails FSD
            WHERE NOT EXISTS (
                SELECT 1 FROM SalaryDetails SD WITH (NOLOCK)
                WHERE SD.EmployeeId = FSD.EmployeeId 
                AND SD.MonthNumber = @Month 
                AND SD.Year = @Year 
                AND SD.isdeleted = 0 
                AND SD.IsEnabled = 1
            );
        
            IF @ExistingCount > 0 AND @NewCount = 0
            BEGIN
                SELECT 'Salary has already been processed for the selected month and year.' AS ResponseMessage, 
                       0 AS Success;
            END
            ELSE
            BEGIN
                SELECT * FROM dbo.AttendanceSummaryTemp att 
                WHERE NOT EXISTS (
                    SELECT 1 FROM SalaryDetails SD WITH (NOLOCK)
                    WHERE SD.EmployeeId = att.EmployeeId 
                    AND SD.MonthNumber = @Month 
                    AND SD.Year = @Year 
                    AND SD.isdeleted = 0 
                    AND SD.IsEnabled = 1
                );
                
                IF @ExistingCount > 0
                BEGIN
                    SELECT CONCAT('Showing ', @NewCount, ' new record(s). ', @ExistingCount, ' record(s) already exist in the database.') AS InfoMessage;
                END
            END
        END
        ELSE IF @Action = 'Insert'
        BEGIN
            INSERT INTO SalaryDetails (
                EmployeeId, EmployeeCode, EmployeeName, GrossSalary, PayableDays, MonthDays, Holiday, AbsentDays, PresentDays,
                WeekOff, Leave, SalaryDays, MonthNumber, MonthName, Year, BasicSalary, HRA, ConveyanceAllowance, ChildEducationAllowance,
                MedicalAllowance, DeputationAllowance, Arrears, TotalGrossSalary, PF, ESIC, ProfessionalTax, GroupMedical, TermInsurance,
                LWF, TDS, Loan, OtherDeduction, TotalDeductions, NetSalary, isEnabled, isDeleted, CreatedBy
            )
            SELECT FSD.EmployeeId, FSD.EmployeeCode, FSD.EmployeeName, FSD.GrossSalary, FSD.PayableDays, FSD.MonthDays, FSD.Holiday,
                   FSD.AbsentDays, FSD.PresentDays, FSD.WeekOff, FSD.Leave, FSD.SalaryDays, FSD.MonthNumber, FSD.MonthName, FSD.Year,
                   FSD.BasicSalary, FSD.HRA, FSD.ConveyanceAllowance, FSD.ChildEducationAllowance, FSD.MedicalAllowance, FSD.DeputationAllowance,
                   FSD.Arrears, FSD.TotalGrossSalary, FSD.PF, FSD.ESIC, FSD.ProfessionalTax, FSD.GroupMedical, FSD.TermInsurance,
                   FSD.LWF, FSD.TDS, FSD.Loan, FSD.OtherDeduction, FSD.TotalDeductions, FSD.NetSalary, 1, 0, @CreatedBy
            FROM #FinalSalaryDetails FSD
            WHERE NOT EXISTS (
                SELECT 1 FROM SalaryDetails SD WITH (NOLOCK)
                WHERE SD.EmployeeId = FSD.EmployeeId AND SD.MonthNumber = @Month 
                AND SD.Year = @Year 
                AND SD.isdeleted = 0 AND SD.IsEnabled = 1
            );
            
            IF @@ROWCOUNT > 0
                SELECT 'Data inserted successfully.' AS ResponseMessage, 1 as Success;
            ELSE
                SELECT 'Data already exists in SalaryDetails table for the given month and year.' AS ResponseMessage, 0 as Success;
        END

        -- Cleanup
        DROP TABLE #FinalSalaryDetails;
        DROP TABLE #AttendanceSummary;
        DROP TABLE #EarningData;
        DROP TABLE #DeductionData;
        DROP TABLE #AttendanceData;
        DROP TABLE #Employees;
    END
    ELSE
    BEGIN
        SELECT 'Invalid Action parameter. Use ''GetData'' or ''Insert''.' AS ResponseMessage, 0 as Success;
    END
END
