                SET @Success = 0;

                SET @ResponseMessage = 'LoginAlias cannot be null or empty.';
                ROLLBACK TRANSACTION;

                SET @Success = 0;
                SET @ResponseMessage = 'EmployeeCode already exists.';

                SET @ResponseMessage = 'LoginAlias already exists.';
                ROLLBACK TRANSACTION;
                RETURN;

            IF NOT EXISTS (SELECT 1 FROM HRMSUserRoles WHERE EmployeeId = @Id AND IsDeleted = 0 AND IsEnabled = 1)
            BEGIN
                EXEC ManageHRMSUserRole
                    @Action = 'CREATE',
                    @EmployeeId = @Id,
                    @RoleId = @RoleId,
                    @CompanyId = @CompanyId,
                    @IsEnabled = @IsEnabled,
                    @IsDeleted = @IsDeleted;
            END
            ELSE
            BEGIN
                EXEC ManageHRMSUserRole
                    @Action = 'UPDATE',
                    @EmployeeId = @Id,
                    @RoleId = @RoleId;
            END

            IF NOT EXISTS (
                SELECT 1
                FROM EmployeeSalaryAllowance
                WHERE EmployeeId = @Id
                    AND IsDeleted = 0
                    AND IsEnabled = 1
            )
            BEGIN
                EXEC USP_CalculateSalaryStructure
                    @Action = 'CREATE',
                    @EmployeeId = @Id,
                    @CompanyId = @CompanyId,
                    @GrossSalary = @GrossSalary,
                    @BasicSalary = @BasicSalary,
                    @IsPFApplicable = @IsPFApplicable;
            END
            ELSE
            BEGIN
                EXEC USP_CalculateSalaryStructure
                    @Action = 'UPDATE',
                    @EmployeeId = @Id,
                    @CompanyId = @CompanyId,
                    @GrossSalary = @GrossSalary,
                    @BasicSalary = @BasicSalary,
                    @IsPFApplicable = @IsPFApplicable;
            END

            DECLARE @LastReportingManagerId INT;
            SELECT TOP 1 @LastReportingManagerId = ReportingManagerId
            FROM ReportingManagerDetails
            WHERE EmployeeId = @Id AND IsDeleted = 0 AND IsEnabled = 1
            ORDER BY ReportingManagerDetailsId DESC;

            IF (@ReportingManagerId <> @LastReportingManagerId)
            BEGIN
                INSERT INTO ReportingManagerDetails (
                    EmployeeId, ReportingManagerId, EffectedDate, IsDeleted, IsEnabled, MethodName, CreatedDate, CreatedBy
                )
                VALUES (
                    @Id, @ReportingManagerId, GETUTCDATE(), 0, 1, 'In Person', GETUTCDATE(), @UpdatedBy
                );
            END

            COMMIT TRANSACTION;
            SET @Success = 1;
            SET @ResponseMessage = 'Basic employee info updated successfully.';
            SELECT @Id AS Id;
        END

        -- UPDATE_PERSONAL: Update personal info
        ELSE IF @Action = 'UPDATE_PERSONAL'
        BEGIN

                DispensaryAddress = ISNULL(@DispensaryAddress, DispensaryAddress),
                UANNumber = ISNULL(@UANNumber, UANNumber),
                DrivingLicense = ISNULL(@DrivingLicense, DrivingLicense),
                DrivingLicenseExpiry = ISNULL(@DrivingLicenseExpiry, DrivingLicenseExpiry),
                RationCardType = ISNULL(@RationCardType, RationCardType),
                RationCardNo = ISNULL(@RationCardNo, RationCardNo),
                ProbationCompletionPeriod = ISNULL(@ProbationCompletionPeriod, ProbationCompletionPeriod),
                ProbationPeriodType = ISNULL(@ProbationPeriodType, ProbationPeriodType),

ALTER PROCEDURE [dbo].[UpdateHRDAttendanceRegularization]
    @AttendanceRegularizationId BIGINT,
    @Status                     NVARCHAR(20),    -- 'ProceedToHRD' or 'Rejected'
    @UpdatedBy                  NVARCHAR(128),
    @Success                    BIT          OUTPUT,
    @ResponseMessage            NVARCHAR(MAX) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Record not found
    IF NOT EXISTS (
        SELECT 1 FROM AttendanceRegularization
        WHERE AttendanceRegularizationId = @AttendanceRegularizationId
          AND IsEnabled = 1
          AND IsDeleted = 0
    )
    BEGIN
        SET @Success         = 0;
        SET @ResponseMessage = 'Record not found.';
        RETURN;
    END

    -- 2. Already approved
    IF @Status = 'ProceedToHRD' AND EXISTS (
        SELECT 1 FROM AttendanceRegularization
        WHERE AttendanceRegularizationId = @AttendanceRegularizationId
          AND proceed_to_HRD = 1
    )
    BEGIN
        SET @Success         = 0;
        SET @ResponseMessage = 'Record is already approved.';
        RETURN;
    END

    -- 3. Already rejected
    IF @Status = 'Rejected' AND EXISTS (
        SELECT 1 FROM AttendanceRegularization
        WHERE AttendanceRegularizationId = @AttendanceRegularizationId
          AND IsRejected = 1
    )
    BEGIN
        SET @Success         = 0;
        SET @ResponseMessage = 'Record is already rejected.';
        RETURN;
    END

    -- 4. Apply update
    UPDATE AttendanceRegularization
    SET
        proceed_to_HRD  = CASE WHEN @Status = 'ProceedToHRD' THEN 1 ELSE 0 END,
        IsPending   = 1,
        IsRejected  = CASE WHEN @Status = 'Rejected'     THEN 1 ELSE 0 END,
        UpdatedBy   = @UpdatedBy,
        UpdatedDate = GETUTCDATE()
    WHERE
        AttendanceRegularizationId = @AttendanceRegularizationId
        AND IsEnabled = 1
        AND IsDeleted = 0;

    SET @Success         = 1;
    SET @ResponseMessage = 'HRD status updated successfully.';
END;

    @IsAllLevelsCompleted BIT OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE
        @PendingStatusId INT = 1,
        @CompleteStatusId INT = 2,
        @HoldStatusId INT = 3,
        @ExtendStatusId INT = 4,
        @RejectStatusId INT = 5;

    SET @Success = 0;
    SET @IsAllLevelsCompleted = 0;

    DECLARE
        @OldStatusId INT,
        @CurrentLevelNo INT,
        @ApprovalLevelId INT,
        @ApprovalMasterId INT,
        @RequesterEmployeeId INT,
        @PrevApproverId INT,
        @LastLevelNo INT,
        @LastApproverEmployeeId INT;

    SELECT
        @OldStatusId = StatusId,
        @CurrentLevelNo = LevelNo,
        @ApprovalLevelId = ApprovalLevelId,
        @PrevApproverId = ApproverEmployeeId
    FROM ApprovalRequestLevel
    WHERE ApprovalRequestLevelId = @ApprovalRequestLevelId
      AND IsActive = 1;

    SELECT
        @ApprovalMasterId = ApprovalMasterId,
        @RequesterEmployeeId = RequesterEmployeeId
    FROM ApprovalRequest
    WHERE ApprovalRequestId = @ApprovalRequestId
      AND IsActive = 1;

    /* ================= ONLY FOR MASTER ID = 1 ================= */
    IF @ApprovalMasterId <> 1
    BEGIN
        SET @Success = 0;
        SET @ResponseMessage = 'This action is allowed only for ApprovalMasterId = 1';
        RETURN;
    END

    /* ================= TERMINAL ACTIONS ================= */
    IF @StatusId IN (@HoldStatusId, @ExtendStatusId, @RejectStatusId)
    BEGIN
        UPDATE ApprovalRequestLevel
        SET StatusId=@StatusId, ActionBy=@ActionBy, ActionOn=GETUTCDATE(),
            ActionRemarks=@Remarks, UpdatedBy=@ActionBy, UpdatedDate=GETUTCDATE()
        WHERE ApprovalRequestLevelId=@ApprovalRequestLevelId;

        UPDATE ApprovalRequest
        SET StatusId=@StatusId, UpdatedBy=@ActionBy,
            UpdatedDate=GETUTCDATE(), ActionDate=GETUTCDATE()
        WHERE ApprovalRequestId=@ApprovalRequestId;

        INSERT INTO ApprovalRequestHistory
        (
            ApprovalRequestId, ApprovalRequestLevelId,
            ActionType, ActionBy, ActionOn,
            Remarks, OldStatusId, NewStatusId,
            CreatedBy, CreatedDate
        )
        VALUES
        (
            @ApprovalRequestId, @ApprovalRequestLevelId,
            'Terminal', @ActionBy, GETUTCDATE(),
            @Remarks, @OldStatusId, @StatusId,
            @ActionBy, GETUTCDATE()
        );

        SET @Success = 1;
        SET @IsAllLevelsCompleted = 1;
        SET @ResponseMessage = 'Request processed successfully.';
        RETURN;
    END

    /* ================= APPROVE ================= */
    IF @StatusId = @CompleteStatusId
    BEGIN
        UPDATE ApprovalRequestLevel
        SET StatusId=@CompleteStatusId, ActionBy=@ActionBy,
            ActionOn=GETUTCDATE(), ActionRemarks=@Remarks,
            UpdatedBy=@ActionBy, UpdatedDate=GETUTCDATE()
        WHERE ApprovalRequestLevelId=@ApprovalRequestLevelId;

        INSERT INTO ApprovalRequestHistory
        (
            ApprovalRequestId, ApprovalRequestLevelId,
            ActionType, ActionBy, ActionOn,
            Remarks, OldStatusId, NewStatusId,
            CreatedBy, CreatedDate
        )
        VALUES
        (
            @ApprovalRequestId, @ApprovalRequestLevelId,
            'Approved', @ActionBy, GETUTCDATE(),
            ISNULL(@Remarks,'Approved'),
            @OldStatusId, @CompleteStatusId,
            @ActionBy, GETUTCDATE()
        );

        DECLARE
            @NextLevelNo INT = @CurrentLevelNo,
            @NextApprovalLevelId INT,
            @ApproverEmployeeId INT,
            @EscalationDays INT,
            @IsDepartmentBased BIT,
            @IsNotMandatory BIT,
            @DepartmentId INT,
            @ShouldSkip BIT;

        SELECT @DepartmentId = DepartmentId
        FROM AspNetUsers
        WHERE Id = @RequesterEmployeeId;

        WHILE 1 = 1
        BEGIN
            SET @ShouldSkip = 0;
            SET @ApproverEmployeeId = NULL;

            SELECT TOP 1
                @NextApprovalLevelId = al.ApprovalLevelId,
                @NextLevelNo = al.LevelNo,
                @EscalationDays = al.EscalationDays,
                @IsDepartmentBased = al.IsDepartmentBased
            FROM ApprovalLevel al
            WHERE al.ApprovalMasterId = @ApprovalMasterId
              AND al.LevelNo > @NextLevelNo
              AND al.IsActive = 1
            ORDER BY al.LevelNo;

            /* ================= LAST LEVEL ================= */
            IF @NextApprovalLevelId IS NULL
            BEGIN
                SET @LastLevelNo = @NextLevelNo;
                SET @LastApproverEmployeeId = @PrevApproverId;

                /* ===== GENERATE REF NO ===== */
                DECLARE @RefNo NVARCHAR(50);
                DECLARE @Year NVARCHAR(4) = CAST(YEAR(GETDATE()) AS NVARCHAR(4));
                DECLARE @NextNo INT;

                SELECT @NextNo = ISNULL(MAX(CAST(RIGHT(ReferenceNo,3) AS INT)),0) + 1
                FROM ApprovalRequest WITH (UPDLOCK, HOLDLOCK)
                WHERE ApprovalMasterId = 1
                  AND ReferenceNo IS NOT NULL
                  AND LEFT(ReferenceNo,4) = @Year;

                SET @RefNo = @Year + '/' + RIGHT('000' + CAST(@NextNo AS NVARCHAR),3);

                UPDATE ApprovalRequest
                SET StatusId=@CompleteStatusId,
                    CurrentLevelNo=@LastLevelNo,
                    CurrentApprover=@LastApproverEmployeeId,
                    ReferenceNo=@RefNo,
                    UpdatedBy=@ActionBy,
                    UpdatedDate=GETUTCDATE(),
                    ActionDate=GETUTCDATE()
                WHERE ApprovalRequestId=@ApprovalRequestId;

                INSERT INTO ApprovalRequestHistory
                (
                    ApprovalRequestId,
                    ActionType, ActionBy, ActionOn,
                    Remarks, OldStatusId, NewStatusId,
                    CreatedBy, CreatedDate
                )
                VALUES
                (
                    @ApprovalRequestId,
                    'Completed', @ActionBy, GETUTCDATE(),
                    'All approval levels completed',
                    @OldStatusId, @CompleteStatusId,
                    @ActionBy, GETUTCDATE()
                );

                SET @IsAllLevelsCompleted = 1;
                SET @Success = 1;
                SET @ResponseMessage = 'Approval completed successfully.';
                RETURN;
            END

            /* ================= NORMAL FLOW ================= */
            IF @IsDepartmentBased = 1
            BEGIN
                SELECT TOP 1
                    @ApproverEmployeeId = ald.ApproverEmployeeId,
                    @IsNotMandatory = ald.IsNotMandatory
                FROM ApprovalLevelDepartment ald
                WHERE ald.ApprovalLevelId = @NextApprovalLevelId
                  AND ald.DepartmentId = @DepartmentId
                  AND ald.IsActive = 1;

                IF @ApproverEmployeeId IS NULL AND ISNULL(@IsNotMandatory,0)=1
                    SET @ShouldSkip = 1;
            END
            ELSE
            BEGIN
                SELECT @ApproverEmployeeId = ApproverEmployeeId
                FROM ApprovalLevel
                WHERE ApprovalLevelId = @NextApprovalLevelId;
            END

            IF @ApproverEmployeeId = @PrevApproverId
                SET @ShouldSkip = 1;

            IF @ShouldSkip = 1
                CONTINUE;

            INSERT INTO ApprovalRequestLevel
            (
                ApprovalRequestId, ApprovalLevelId, LevelNo,
                ApproverEmployeeId, StatusId, IsActive,
                CreatedBy, CreatedDate, AssignedOn, EscalationDueOn
            )
            VALUES
            (
                @ApprovalRequestId, @NextApprovalLevelId, @NextLevelNo,
                @ApproverEmployeeId, @PendingStatusId, 1,
                @ActionBy, GETUTCDATE(), GETUTCDATE(),
                DATEADD(DAY, ISNULL(@EscalationDays,0), GETUTCDATE())
            );

            UPDATE ApprovalRequest
            SET CurrentLevelNo = @NextLevelNo,
                CurrentApprover = @ApproverEmployeeId,
                UpdatedBy = @ActionBy,
                UpdatedDate = GETUTCDATE()
            WHERE ApprovalRequestId = @ApprovalRequestId;

            SET @Success = 1;
            SET @ResponseMessage = 'Approved and next level assigned.';
            RETURN;
        END
    END
END

        emp.EmployeeCode,
		emp.Initial,
    emp.FirstName,
    emp.MiddleName,
    emp.LastName,
        emp.FullName AS EmployeeName,
        emp.PersonalEmailId,
        cd.CompanyName,
		b.BranchName,

        ar.ApprovalRequestId,

        -- ✅ SAME alias
        ISNULL(ar.ReferenceNo, 'Pending') AS ReferenceNo,

        emp.DateOfJoining,
        emp.ProbationEndDate,
        emp.ProbationDueDate,

        ar.StatusId,
        ps.ProbationStatusName AS StatusName,
        ar.ActionDate,

        ar.IsMailSent,

        reporting.EmployeeCode AS ReportingManagerCode,
        reporting.FullName AS ReportingManagerName,

        ISNULL(
            CAST(pep.ProbationEvaluationDays AS VARCHAR(10)) + ' Days',
            'N/A'

    INNER JOIN ProbationStatus ps
        ON ps.ProbationStatusId = ar.StatusId
       AND ps.IsActive = 1
    INNER JOIN AspNetUsers emp
        ON emp.Id = ar.RequesterEmployeeId
    LEFT JOIN AspNetUsers reporting
        ON reporting.Id = emp.ReportingManagerId
    LEFT JOIN CompanyDetails cd
        ON cd.CompanyId = emp.CompanyId
    LEFT JOIN AspNetUsers currentApprover
        ON currentApprover.Id = ar.CurrentApprover
    LEFT JOIN ProbationPerformance pp
        ON pp.ApprovalRequestId = ar.ApprovalRequestId
       AND pp.LevelNo = ar.CurrentLevelNo
       AND pp.IsActive = 1
    LEFT JOIN ProbationEvaluationPeriod pep
        ON pep.ProbationEvaluationPeriodId = pp.ProbationEvaluationPeriodId
		LEFT JOIN Branch b
    ON b.BranchId = emp.BranchId
    WHERE ar.IsActive = 1
      AND emp.CompanyId = @CompanyId
      AND (@StatusId IS NULL OR ar.StatusId = @StatusId)
      AND (@EmployeeId IS NULL OR emp.Id = @EmployeeId)
      AND (
            @Year IS NULL
            OR (
                emp.DateOfJoining >= DATEFROMPARTS(@Year, 1, 1)
                AND emp.DateOfJoining < DATEFROMPARTS(@Year + 1, 1, 1)
            )
          )
      AND ar.ApprovalMasterId = @ApprovalMasterId;

    SELECT
        mr.*,

            SELECT

            LEFT JOIN AspNetUsers u
                ON u.Id = arl.ApproverEmployeeId

                ON ps2.ProbationStatusId = arl.StatusId
            LEFT JOIN ProbationPerformance pp
                ON pp.ApprovalRequestLevelId = arl.ApprovalRequestLevelId
               AND pp.IsActive = 1

            WHERE arl.ApprovalRequestId = mr.ApprovalRequestId
              AND (arl.LevelNo < mr.CurrentLevelNo OR mr.CurrentLevelNo IS NULL)
              AND arl.IsActive = 1

            ORDER BY arl.LevelNo
            FOR JSON PATH
        ) AS ApprovalHistoryJson

    FROM #MainRequests mr
    ORDER BY mr.ActionDate DESC;

    DROP TABLE #MainRequests;
END

ALTER PROCEDURE [dbo].[usp_EscalatePendingApprovalRequests]
(
    @Success BIT = NULL OUTPUT,
    @ResponseMessage NVARCHAR(MAX) = NULL OUTPUT,
    @EscalatedData NVARCHAR(MAX) = NULL OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Now DATETIME = GETUTCDATE();
    DECLARE @PendingStatusId INT = 1;
    DECLARE @EscalateStatusId INT = 6;

    SET @Success = 1;
    SET @ResponseMessage = '';
    SET @EscalatedData = NULL;

    CREATE TABLE #EscalatedRequests
    (
        ApprovalRequestId INT,
        ApprovalRequestLevelId INT,
        LevelNo INT
    );

    /* ================= ESCALATE (ONLY IF NEXT LEVEL EXISTS) ================= */
    UPDATE arl
    SET
        arl.StatusId = @EscalateStatusId,
        arl.EscalatedOn = @Now,
        arl.EscalationDueOn = DATEADD(DAY, 1, @Now),
        arl.UpdatedBy = 0,
        arl.UpdatedDate = @Now
    OUTPUT
        inserted.ApprovalRequestId,
        inserted.ApprovalRequestLevelId,
        inserted.LevelNo
    INTO #EscalatedRequests
    FROM dbo.ApprovalRequestLevel arl
    INNER JOIN dbo.ApprovalRequest ar 
        ON ar.ApprovalRequestId = arl.ApprovalRequestId
    WHERE
        arl.StatusId = @PendingStatusId
        AND arl.IsActive = 1
        AND arl.EscalationDueOn <= @Now
        AND ar.ApprovalMasterId = 1
        AND EXISTS (
            SELECT 1
            FROM ApprovalLevel al
            WHERE al.ApprovalMasterId = ar.ApprovalMasterId
              AND al.LevelNo > arl.LevelNo
              AND al.IsActive = 1
        );

    /* ================= HISTORY ================= */
    INSERT INTO dbo.ApprovalRequestHistory
    (
        ApprovalRequestId,
        ApprovalRequestLevelId,
        ActionType,
        ActionBy,
        Remarks,
        OldStatusId,
        NewStatusId,
        CreatedDate
    )
    SELECT
        ApprovalRequestId,
        ApprovalRequestLevelId,
        'Escalated',
        1,
        'Request escalated due to inaction',
        @PendingStatusId,
        @EscalateStatusId,
        @Now
    FROM #EscalatedRequests;

    /* ================= VARIABLES ================= */
    DECLARE
        @RequestId INT,
        @LevelId INT,
        @LevelNo INT,
        @ApprovalMasterId INT,
        @ApprovalLevelId INT,
        @ApproverEmployeeId INT,
        @EscalationDays INT,
        @DepartmentId INT,
        @IsNotMandatory BIT,
        @ShouldSkip BIT,
        @PrevApproverId INT;

    /* ================= CURSOR ================= */
    DECLARE EscalatedCursor CURSOR FOR
        SELECT ApprovalRequestId, ApprovalRequestLevelId, LevelNo
        FROM #EscalatedRequests;

    OPEN EscalatedCursor;
    FETCH NEXT FROM EscalatedCursor INTO @RequestId, @LevelId, @LevelNo;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        SELECT @PrevApproverId = ApproverEmployeeId
        FROM ApprovalRequestLevel
        WHERE ApprovalRequestLevelId = @LevelId;

        SELECT @ApprovalMasterId = ApprovalMasterId
        FROM ApprovalRequest
        WHERE ApprovalRequestId = @RequestId
          AND IsActive = 1;

        SELECT @DepartmentId = DepartmentId
        FROM AspNetUsers
        WHERE Id = (
            SELECT RequesterEmployeeId
            FROM ApprovalRequest
            WHERE ApprovalRequestId = @RequestId
        );

        SELECT TOP 1
            @ApprovalLevelId = al.ApprovalLevelId,
            @EscalationDays = al.EscalationDays
        FROM ApprovalLevel al
        WHERE al.ApprovalMasterId = @ApprovalMasterId
          AND al.LevelNo > @LevelNo
          AND al.IsActive = 1
        ORDER BY al.LevelNo;

        WHILE @ApprovalLevelId IS NOT NULL
        BEGIN
            SET @ShouldSkip = 0;
            SET @ApproverEmployeeId = NULL;
            SET @IsNotMandatory = 0;

            IF EXISTS (
                SELECT 1
                FROM ApprovalLevel
                WHERE ApprovalLevelId = @ApprovalLevelId
                  AND IsDepartmentBased = 1
            )
            BEGIN
                SELECT TOP 1
                    @ApproverEmployeeId = ald.ApproverEmployeeId,
                    @IsNotMandatory = ald.IsNotMandatory
                FROM ApprovalLevelDepartment ald
                WHERE ald.ApprovalLevelId = @ApprovalLevelId
                  AND ald.DepartmentId = @DepartmentId
                  AND ald.IsActive = 1;

                IF @ApproverEmployeeId IS NULL AND ISNULL(@IsNotMandatory, 0) = 1
                    SET @ShouldSkip = 1;
            END

            IF @ApproverEmployeeId = @PrevApproverId
                SET @ShouldSkip = 1;

            IF @ShouldSkip = 1
            BEGIN
                SET @LevelNo = @LevelNo + 1;

                SELECT TOP 1
                    @ApprovalLevelId = al.ApprovalLevelId,
                    @EscalationDays = al.EscalationDays
                FROM ApprovalLevel al
                WHERE al.ApprovalMasterId = @ApprovalMasterId
                  AND al.LevelNo > @LevelNo
                  AND al.IsActive = 1
                ORDER BY al.LevelNo;

                CONTINUE;
            END

            IF @ApproverEmployeeId IS NULL
            BEGIN
                SELECT @ApproverEmployeeId = ApproverEmployeeId
                FROM ApprovalLevel
                WHERE ApprovalLevelId = @ApprovalLevelId;
            END

            IF EXISTS (
                SELECT 1 FROM ApprovalLevel
                WHERE ApprovalLevelId = @ApprovalLevelId
                  AND IsReportingPerson = 1
            )
            BEGIN
                SELECT @ApproverEmployeeId = ReportingManagerId
                FROM AspNetUsers
                WHERE Id = (
                    SELECT RequesterEmployeeId
                    FROM ApprovalRequest
                    WHERE ApprovalRequestId = @RequestId
                );
            END

            INSERT INTO ApprovalRequestLevel
            (
                ApprovalRequestId,
                ApprovalLevelId,
                LevelNo,
                ApproverEmployeeId,
                StatusId,
                IsActive,
                CreatedBy,
                CreatedDate,
                AssignedOn,
                EscalationDueOn
            )
            VALUES
            (
                @RequestId,
                @ApprovalLevelId,
                @LevelNo + 1,
                @ApproverEmployeeId,
                @PendingStatusId,
                1,
                1,
                @Now,
                @Now,
                DATEADD(DAY, ISNULL(@EscalationDays, 0), @Now)
            );

            DECLARE @NewLevelId INT = SCOPE_IDENTITY();

            UPDATE ApprovalRequest
            SET
                CurrentLevelNo = @LevelNo + 1,
                CurrentApprover = @ApproverEmployeeId,
                UpdatedDate = @Now,
                ActionDate = @Now
            WHERE ApprovalRequestId = @RequestId;

            INSERT INTO ApprovalRequestHistory
            (
                ApprovalRequestId,
                ApprovalRequestLevelId,
                ActionType,
                ActionBy,
                Remarks,
                OldStatusId,
                NewStatusId,
                CreatedDate
            )
            VALUES
            (
                @RequestId,
                @NewLevelId,
                'Assigned',
                1,
                'Assigned after escalation',
                @EscalateStatusId,
                @PendingStatusId,
                @Now
            );

            BREAK;
        END

        FETCH NEXT FROM EscalatedCursor INTO @RequestId, @LevelId, @LevelNo;
    END

    CLOSE EscalatedCursor;
    DEALLOCATE EscalatedCursor;

    /* ================= JSON OUTPUT ================= */
    IF EXISTS (SELECT 1 FROM #EscalatedRequests)
    BEGIN
        SELECT @EscalatedData =
        (
            SELECT
                ar.RequesterEmployeeId AS EmployeeId,
                emp.FullName AS EmployeeName,
                emp.EmployeeCode AS EmployeeCode,
                prev.OfficialEmail AS PreviousApproverEmail,
                prev.FullName AS PreviousApprover,
                prev.EmployeeCode AS PreviousApproverCode,
                arl.LevelNo AS PreviousLevelNo,
                nextAppr.OfficialEmail AS NextApproverEmail,
                nextAppr.FullName AS NextApprover,
                nextAppr.EmployeeCode AS NextApproverCode,
                nl.LevelNo AS NextLevelNo,
                c.CompanyName,
                arl.EscalatedOn AS EscalationDate
            FROM #EscalatedRequests er
            JOIN ApprovalRequest ar ON ar.ApprovalRequestId = er.ApprovalRequestId
            JOIN ApprovalRequestLevel arl ON arl.ApprovalRequestLevelId = er.ApprovalRequestLevelId
            LEFT JOIN AspNetUsers emp ON emp.Id = ar.RequesterEmployeeId
            LEFT JOIN AspNetUsers prev ON prev.Id = arl.ApproverEmployeeId
            LEFT JOIN ApprovalRequestLevel nl ON nl.ApprovalRequestId = ar.ApprovalRequestId
                                             AND nl.LevelNo = ar.CurrentLevelNo
            LEFT JOIN AspNetUsers nextAppr ON nextAppr.Id = nl.ApproverEmployeeId
            LEFT JOIN CompanyDetails c ON c.CompanyId = emp.CompanyId
            FOR JSON PATH, INCLUDE_NULL_VALUES
        );
    END

    DROP TABLE #EscalatedRequests;

    SET @ResponseMessage = 'Escalation completed (last level untouched)';
END

ALTER PROCEDURE [dbo].[usp_AutomateProbationEndApprovalRequests]
(
    @ApprovalMasterId INT,
    @StartDate DATE = NULL,
    @EndDate DATE = NULL,
    @Success BIT = NULL OUTPUT,
    @ResponseMessage NVARCHAR(MAX) = NULL OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Now DATETIME = GETUTCDATE();

    /* ===== DEFAULT DATE LOGIC ===== */
    IF @StartDate IS NULL OR @EndDate IS NULL
    BEGIN
        SET @StartDate = CAST(@Now AS DATE);
        SET @EndDate = DATEADD(DAY, 15, @StartDate);
    END

    DECLARE 
        @EmployeeId INT,
        @EmployeeName NVARCHAR(200),
        @ApprovalRequestId INT,
        @RequestTitle NVARCHAR(250) = 'Probation End Approval Request',
        @CreatedBy INT = 0,
        @PendingStatusId INT = 1;

    SET @Success = 1;
    SET @ResponseMessage = 'Automation started';

    DECLARE EmployeeCursor CURSOR FOR
        SELECT Id, FullName + ' - ' + EmployeeCode
        FROM AspNetUsers
        WHERE
            IsEnabled = 1
            AND IsDeleted = 0
            AND ISNULL(IsLeft, 0) = 0
            AND Probation = 1
            AND ProbationEndDate BETWEEN @StartDate AND @EndDate;

    OPEN EmployeeCursor;
    FETCH NEXT FROM EmployeeCursor INTO @EmployeeId, @EmployeeName;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        IF NOT EXISTS (
            SELECT 1 FROM ApprovalRequest
            WHERE RequesterEmployeeId = @EmployeeId
              AND ApprovalMasterId = @ApprovalMasterId
              AND IsActive = 1
        )
        BEGIN
            INSERT INTO ApprovalRequest
            (
                ApprovalMasterId, RequesterEmployeeId, RequestTitle,
                CurrentLevelNo, StatusId, IsActive,
                CreatedBy, CreatedDate, IsMailSent
            )
            VALUES
            (
                @ApprovalMasterId, @EmployeeId, @RequestTitle,
                1, @PendingStatusId, 1,
                @CreatedBy, @Now, 0
            );

            SET @ApprovalRequestId = SCOPE_IDENTITY();

            /* ===== FIRST LEVEL ===== */
            DECLARE 
                @ApprovalLevelId INT,
                @ApproverEmployeeId INT,
                @EscalationDays INT,
                @IsDepartmentBased BIT,
                @DepartmentId INT;

            SELECT TOP 1
                @ApprovalLevelId = al.ApprovalLevelId,
                @ApproverEmployeeId = al.ApproverEmployeeId,
                @EscalationDays = al.EscalationDays,
                @IsDepartmentBased = al.IsDepartmentBased
            FROM ApprovalLevel al
            WHERE al.ApprovalMasterId = @ApprovalMasterId
              AND al.LevelNo = 1
              AND al.IsActive = 1;

            /* ===== GET EMPLOYEE DEPARTMENT ===== */
            SELECT @DepartmentId = DepartmentId
            FROM AspNetUsers
            WHERE Id = @EmployeeId;

            /* ===== APPROVER LOGIC ===== */
            IF @IsDepartmentBased = 1
            BEGIN
                SELECT @ApproverEmployeeId = ald.ApproverEmployeeId
                FROM ApprovalLevelDepartment ald
                WHERE ald.ApprovalLevelId = @ApprovalLevelId
                  AND ald.DepartmentId = @DepartmentId
                  AND ald.IsActive = 1;
            END
            ELSE IF EXISTS (
                SELECT 1 FROM ApprovalLevel
                WHERE ApprovalLevelId = @ApprovalLevelId
                  AND IsReportingPerson = 1
            )
            BEGIN
                SELECT @ApproverEmployeeId = ReportingManagerId
                FROM AspNetUsers
                WHERE Id = @EmployeeId;
            END

            /* ===== FALLBACK SAFETY ===== */
            IF @ApproverEmployeeId IS NULL
            BEGIN
                SELECT @ApproverEmployeeId = ApproverEmployeeId
                FROM ApprovalLevel
                WHERE ApprovalLevelId = @ApprovalLevelId;
            END

            /* ===== UPDATE CURRENT APPROVER ===== */
            UPDATE ApprovalRequest
            SET CurrentApprover = @ApproverEmployeeId
            WHERE ApprovalRequestId = @ApprovalRequestId;

            /* ===== INSERT LEVEL ===== */
            INSERT INTO ApprovalRequestLevel
            (
                ApprovalRequestId, ApprovalLevelId, LevelNo,
                ApproverEmployeeId, StatusId, IsActive,
                CreatedBy, CreatedDate, AssignedOn, EscalationDueOn
            )
            VALUES
            (
                @ApprovalRequestId, @ApprovalLevelId, 1,
                @ApproverEmployeeId, @PendingStatusId, 1,
                @CreatedBy, @Now, @Now,
                DATEADD(DAY, ISNULL(@EscalationDays,0), @Now)
            );

            /* ===== HISTORY ===== */
            INSERT INTO ApprovalRequestHistory
            (
                ApprovalRequestId, ActionType, ActionBy,
                Remarks, NewStatusId, CreatedDate
            )
            VALUES
            (
                @ApprovalRequestId, 'Inserted', @CreatedBy,
                CONCAT('Approval request created for ', @EmployeeName),
                @PendingStatusId, @Now
            );
        END

        FETCH NEXT FROM EmployeeCursor INTO @EmployeeId, @EmployeeName;
    END

    CLOSE EmployeeCursor;
    DEALLOCATE EmployeeCursor;

    SET @ResponseMessage = CONCAT('Completed from ', @StartDate, ' to ', @EndDate);
END

        AND ReferenceNo IS NULL
)
UPDATE ar
SET ReferenceNo = 
    @Year + '/' + RIGHT('000' + CAST(cte.RowNum AS NVARCHAR), 3)
FROM ApprovalRequest ar
INNER JOIN CTE cte
    ON ar.ApprovalRequestId = cte.ApprovalRequestId;
